async function safeSelect(tableName, orderCol) {
  try {
    const res = await supabase.from(tableName).select('*').limit(500);

}

function getFiltered() {
  return presupuestos.filter(p=>{
    if(filterEstadoVal!=='todos' && p.estado!==filterEstadoVal) return false;
    const s=(document.getElementById('searchPres')?.value||'').toLowerCase();
    if(s) return (p.numero||'').toLowerCase().includes(s)||(p.cliente||'').toLowerCase().includes(s)||(p.descripcion||'').toLowerCase().includes(s);
    return true;
  }).sort((a,b)=>(b.numero||'').localeCompare(a.numero||''));
}

async function confirmarEliminar(id) {

  if(res.error){ toast(res.error.message,'err'); return; }

}

function renderLines() {
  return modalLines.map((l,i)=>`<div class="line-row">
    <input class="form-input" value="${esc(l.desc||'')}" onchange="updateLine(${i},'desc',this.value)" placeholder="Descripcion">
    <input class="form-input" type="number" min="1" step="1" value="${l.qty||''}" onchange="updateLine(${i},'qty',parseFloat(this.value)||1)" style="font-family:var(--font-mono)">
    <input class="form-input" type="number" min="0" step="0.01" value="${l.price||''}" onchange="updateLine(${i},'price',parseFloat(this.value)||0)" placeholder="0.00" style="font-family:var(--font-mono)">
    <input class="form-input" type="number" readonly value="${lineaTotal(l).toFixed(2)}" style="font-family:var(--font-mono);color:var(--blue2)">
    <button class="btn btn-icon" style="color:var(--red);" onclick="removeLine(${i})" ${modalLines.length<=1?'disabled':''}>X</button>
  </div>`).join('');
}

function updateLine(i,field,val) {
  modalLines = modalLines.map((l,idx)=>{
    if(idx!==i) return l;
    const u = Object.assign({}, l);
    u[field]=val;
    return u;
  });
  renderModal();
}

function addLine() { modalLines = modalLines.concat([{desc:'',qty:1,price:0}]); renderModal(); }

function removeLine(i) { if(modalLines.length<=1) return; modalLines = modalLines.filter((_,idx)=>idx!==i); renderModal(); }

function autoCliente(name) {
  const c = clientes.find(x=>x.nombre===name);
  if(c) {
    if(!modalForm.direccion) modalForm.direccion = c.direccion||'';
    if(!modalForm.telefono) modalForm.telefono = c.telefono||'';
    renderModal();
  }
}

function cerrarModal() { modalOpen=false; editingPres=null; templateMode=null; renderModal(); }

async function guardarPresupuesto() {
  if(!modalForm.cliente || !modalForm.cliente.trim()){ toast('El cliente es obligatorio','err'); return; }
  const baseImponible = modalLines.reduce((s,l)=>s+lineaTotal(l),0);
  const ivaPct = Number(modalForm.iva_pct)||0;
  const data = Object.assign({}, modalForm, {
    base_imponible: baseImponible,
    iva_importe: baseImponible*(ivaPct/100),
    total: baseImponible*(1+(ivaPct/100)),
    lineas: JSON.stringify(modalLines),
    descripcion: buildDescripcion(modalLines)
  });
  let error;
  if(editingPres) {
    const res = await supabase.from('presupuestos').update(data).eq('id',editingPres.id);
    error = res.error;
    if(!error && data.estado==='Aceptado'){
      try {
        const exist = trabajos.find(t=>t.presupuesto_id===editingPres.id);
        if(!exist) await supabase.from('trabajos').insert({presupuesto_id:editingPres.id,presupuesto_num:data.numero,cliente:data.cliente,descripcion:data.descripcion,base_imponible:data.base_imponible,iva_pct:data.iva_pct,iva_importe:data.iva_importe,total:data.total,estado:'Pendiente',tecnico:'Jesus (T1)',enviado_contable:false});
      } catch(ee) { console.warn('No se pudo crear trabajo:',ee.message); }
    }
  } else {
    const res = await supabase.from('presupuestos').insert(data).select();
    error = res.error;
    if(!error && data.estado==='Aceptado' && res.data && res.data[0]) {
      try { await supabase.from('trabajos').insert({presupuesto_id:res.data[0].id,presupuesto_num:data.numero,cliente:data.cliente,descripcion:data.descripcion,base_imponible:data.base_imponible,iva_pct:data.iva_pct,iva_importe:data.iva_importe,total:data.total,estado:'Pendiente',tecnico:'Jesus (T1)',enviado_contable:false}); } catch(ee) { console.warn('No se pudo crear trabajo:',ee.message); }
    }
  }
  if(error){ toast(error.message,'err'); return; }
  cerrarModal();
  await cargarDatos(); renderPresupuestos();
  toast(editingPres?'Actualizado':'Creado','ok');
}

function importCSV() {
  const input=document.createElement('input'); input.type='file'; input.accept='.csv';
  input.onchange=async function(e){
    const file=e.target.files[0]; if(!file) return;
    const text=await file.text();
    const lines=text.split('\n').filter(l=>l.trim());
    if(lines.length<2) { toast('CSV vacio','err'); return; }
    const parseLine=function(line){
      const r=[]; let c='',q=false;
      for(let i=0;i<line.length;i++){
        const ch=line[i];
        if(ch==='"'){ if(q&&line[i+1]==='"'){c+='"';i++;} else q=!q; }
        else if((ch===','||ch===';')&&!q){ r.push(c.trim()); c=''; }
        else c+=ch;
      }
      r.push(c.trim());
      return r;
    };
    const headers=parseLine(lines[0]).map(h=>h.toLowerCase());
    let count=0;
    for(let i=1;i<lines.length;i++){
      const cols=parseLine(lines[i]);
      const find=function(key){ const idx=headers.findIndex(h=>h.includes(key)); return idx>=0?cols[idx]:''; };
      const estadoRaw=find('estado')||'Pendiente';
      const estado=estadoRaw.charAt(0).toUpperCase()+estadoRaw.slice(1).toLowerCase();
      await supabase.from('presupuestos').insert({
        numero:find('numero')||nextNum(),
        fecha_emision:find('fecha')||today(),
        estado:(['Pendiente','Aceptado','Rechazado'].includes(estado)?estado:'Pendiente'),
        cliente:find('cliente')||'',
        direccion:find('direccion')||'',
        telefono:find('telefono')||'',
        iva_pct:parseFloat(find('iva'))||21,
        base_imponible:parseFloat(find('base'))||0,
        iva_importe:parseFloat(find('cuota'))||0,
        total:parseFloat(find('total'))||0,
        descripcion:find('descripcion')||'',
      });
      count++;
    }
    await cargarDatos(); renderPresupuestos(); toast(count+' importados','ok');
  };
  input.click();
}

function downloadFile(content,filename,mime) {
  const a=document.createElement('a');
  a.href=URL.createObjectURL(new Blob([content],{type:mime}));
  a.download=filename; a.click();
}
